-- create table - Description: Analytical Views for Company ESG KPIs and Sector Medians
DROP VIEW IF EXISTS view_sector_benchmarks CASCADE;
DROP VIEW IF EXISTS view_company_esg_kpis CASCADE;

-- ----------------------------------------------------------------------------
-- View 1: 18 ESG KPIs with Statutory Context Priority
-- Preserves ALL registered entities via LEFT JOIN from entity_list
-- ----------------------------------------------------------------------------
CREATE OR REPLACE VIEW view_company_esg_kpis AS
WITH ranked_facts AS (
    SELECT 
        f.client_uuid,
        CASE 
            -- GHG Emissions (Scope 1, 2, 3)
            WHEN f.standard_indicator_capmkt ILIKE '%scope 1%' OR f.indicator_capmkt ILIKE '%scope 1%' THEN 'scope_1_tco2e'
            WHEN f.standard_indicator_capmkt ILIKE '%scope 2%' OR f.indicator_capmkt ILIKE '%scope 2%' THEN 'scope_2_tco2e'
            WHEN f.standard_indicator_capmkt ILIKE '%scope 3%' OR f.indicator_capmkt ILIKE '%scope 3%' THEN 'scope_3_tco2e'
            
            -- Environmental & Resource Intensities
            WHEN f.standard_indicator_capmkt ILIKE '%renewable%' AND f.standard_indicator_capmkt ILIKE '%percent%' THEN 'renewable_energy_pct'
            WHEN f.standard_indicator_capmkt ILIKE '%water%' AND (f.standard_indicator_capmkt ILIKE '%withdrawal%' OR f.standard_indicator_capmkt ILIKE '%consumption%') THEN 'water_withdrawal_kl'
            WHEN f.standard_indicator_capmkt ILIKE '%energy%intensity%' THEN 'energy_intensity'
            WHEN f.standard_indicator_capmkt ILIKE '%waste%intensity%' THEN 'waste_intensity'
            
            -- Social, Diversity & Remuneration
            WHEN (f.standard_indicator_capmkt ILIKE '%female%' OR f.standard_indicator_capmkt ILIKE '%women%') 
                 AND (f.standard_indicator_capmkt ILIKE '%director%' OR f.standard_indicator_capmkt ILIKE '%board%') 
                 THEN 'female_board_directors_pct'
            WHEN f.standard_indicator_capmkt ILIKE '%wages%' AND f.standard_indicator_capmkt ILIKE '%female%' THEN 'gross_wages_to_female_pct'
            WHEN f.standard_indicator_capmkt ILIKE '%turnover%' THEN 'employee_turnover_pct'
            WHEN f.standard_indicator_capmkt ILIKE '%injur%' THEN 'work_related_injuries'
            WHEN f.standard_indicator_capmkt ILIKE '%fatal%' THEN 'fatalities'
            WHEN (f.standard_indicator_capmkt ILIKE '%female%' OR f.standard_indicator_capmkt ILIKE '%women%') 
                 AND f.standard_indicator_capmkt ILIKE '%workforce%' 
                 THEN 'female_workforce_pct'
            
            -- Governance: POSH Specific (Evaluated first to prevent overlap with general complaints)
            WHEN (f.standard_indicator_capmkt ILIKE '%posh%' OR f.indicator_capmkt ILIKE '%posh%')
                 AND (f.standard_indicator_capmkt ILIKE '%upheld%' OR f.indicator_capmkt ILIKE '%upheld%') 
                 THEN 'posh_complaints_upheld'
            WHEN (f.standard_indicator_capmkt ILIKE '%posh%' OR f.indicator_capmkt ILIKE '%posh%') 
                 THEN 'total_posh_complaints'
            
            -- Governance: General Statutory Complaints (Excluding POSH)
            WHEN (f.standard_indicator_capmkt NOT ILIKE '%posh%' AND f.indicator_capmkt NOT ILIKE '%posh%')
                 AND (f.standard_indicator_capmkt ILIKE '%complaint%' OR f.indicator_capmkt ILIKE '%complaint%')
                 AND (f.standard_indicator_capmkt ILIKE '%filed%' OR f.indicator_capmkt ILIKE '%filed%') 
                 THEN 'complaints_filed'
            WHEN (f.standard_indicator_capmkt NOT ILIKE '%posh%' AND f.indicator_capmkt NOT ILIKE '%posh%')
                 AND (f.standard_indicator_capmkt ILIKE '%complaint%' OR f.indicator_capmkt ILIKE '%complaint%')
                 AND (f.standard_indicator_capmkt ILIKE '%pending%' OR f.indicator_capmkt ILIKE '%pending%') 
                 THEN 'pending_complaints'
            
            -- Governance: Disciplinary Actions & Wellbeing Spending
            WHEN f.standard_indicator_capmkt ILIKE '%disciplinary%' THEN 'disciplinary_actions'
            WHEN f.standard_indicator_capmkt ILIKE '%wellbeing%' OR f.standard_indicator_capmkt ILIKE '%well-being%' THEN 'cost_on_wellbeing_pct'
            ELSE NULL
        END AS metric_name,
        COALESCE(f.standard_numeric_value, f.numeric_value::numeric) AS metric_value,
        -- Context Priority: Boundary anchors prevent accidental '%cy%' substring matching
        CASE 
            WHEN f.standard_context_reference ILIKE ANY(ARRAY['%_cy%', '%cy_%', '%dcy%', '%icy%', '%current%', '%main%']) THEN 3
            WHEN f.standard_context_reference ILIKE ANY(ARRAY['%_py%', '%py_%', '%dpy%', '%ipy%', '%previous%', '%prior%']) THEN 1
            ELSE 2
        END AS context_priority
    FROM brsr_facts f
),
resolved_metrics AS (
    -- Resolves the single highest-priority reported value per entity per metric
    SELECT DISTINCT ON (client_uuid, metric_name)
        client_uuid,
        metric_name,
        metric_value
    FROM ranked_facts
    WHERE metric_name IS NOT NULL AND metric_value IS NOT NULL
    ORDER BY client_uuid, metric_name, context_priority DESC, metric_value DESC
)
-- Base table is entity_list: Every company remains visible regardless of metric presence
SELECT 
    e.entity_id,
    e.company_name,
    e.corporate_id_number,
    COALESCE(e.industry_sector, 'Diversified / Other') AS industry_sector,
    COALESCE(e.industry_division, 'General / Unclassified') AS industry_division,
    e.financial_year_start,
    e.financial_year_end,
    ROUND((e.paid_up_shares_value / 10000000.0)::numeric, 2) AS paid_up_capital_cr,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'scope_1_tco2e') AS scope_1_tco2e,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'scope_2_tco2e') AS scope_2_tco2e,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'scope_3_tco2e') AS scope_3_tco2e,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'water_withdrawal_kl') AS water_withdrawal_kl,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'energy_intensity') AS energy_intensity,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'waste_intensity') AS waste_intensity,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'renewable_energy_pct') AS renewable_energy_pct,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'female_board_directors_pct') AS female_board_directors_pct,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'gross_wages_to_female_pct') AS gross_wages_to_female_pct,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'employee_turnover_pct') AS employee_turnover_pct,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'work_related_injuries') AS work_related_injuries,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'fatalities') AS fatalities,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'female_workforce_pct') AS female_workforce_pct,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'posh_complaints_upheld') AS posh_complaints_upheld,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'total_posh_complaints') AS total_posh_complaints,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'complaints_filed') AS complaints_filed,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'pending_complaints') AS pending_complaints,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'disciplinary_actions') AS disciplinary_actions,
    MAX(r.metric_value) FILTER (WHERE r.metric_name = 'cost_on_wellbeing_pct') AS cost_on_wellbeing_pct
FROM entity_list e
LEFT JOIN resolved_metrics r ON e.entity_id = r.client_uuid
GROUP BY 
    e.entity_id, 
    e.company_name, 
    e.corporate_id_number, 
    e.industry_sector, 
    e.industry_division, 
    e.financial_year_start, 
    e.financial_year_end, 
    e.paid_up_shares_value;

-- ----------------------------------------------------------------------------
-- View 2: Native Statistical Medians per Sector
-- Automatically updates when view_company_esg_kpis is queried
-- ----------------------------------------------------------------------------
CREATE OR REPLACE VIEW view_sector_benchmarks AS
SELECT 
    industry_sector,
    COUNT(entity_id) AS entities_reporting,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY scope_1_tco2e)::numeric, 2) AS median_scope_1,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY scope_2_tco2e)::numeric, 2) AS median_scope_2,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY scope_3_tco2e)::numeric, 2) AS median_scope_3,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY water_withdrawal_kl)::numeric, 2) AS median_water,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY energy_intensity)::numeric, 4) AS median_energy_intensity,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY waste_intensity)::numeric, 4) AS median_waste_intensity,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY renewable_energy_pct)::numeric, 2) AS median_renewable_energy_pct,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY female_board_directors_pct)::numeric, 2) AS median_female_board_pct,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY gross_wages_to_female_pct)::numeric, 2) AS median_gross_wages_pct,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY employee_turnover_pct)::numeric, 2) AS median_turnover_pct,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY work_related_injuries)::numeric, 2) AS median_injuries,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY fatalities)::numeric, 2) AS median_fatalities,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY female_workforce_pct)::numeric, 2) AS median_female_workforce_pct,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY posh_complaints_upheld)::numeric, 2) AS median_posh_upheld,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY total_posh_complaints)::numeric, 2) AS median_total_posh,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY complaints_filed)::numeric, 2) AS median_complaints_filed,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY pending_complaints)::numeric, 2) AS median_pending_complaints,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY disciplinary_actions)::numeric, 2) AS median_disciplinary_actions,
    ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY cost_on_wellbeing_pct)::numeric, 2) AS median_cost_wellbeing_pct
FROM view_company_esg_kpis
GROUP BY industry_sector;
