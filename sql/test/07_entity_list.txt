-- create table - Company Master Disclosures & Temporal Boundaries
CREATE TABLE entity_list (
    entity_id UUID PRIMARY KEY DEFAULT uuidv7(),
    section VARCHAR(10) NOT NULL DEFAULT 'A',
    principle_number SMALLINT NOT NULL DEFAULT 0,
    sub_section SMALLINT NOT NULL DEFAULT 1,
    friendly_group VARCHAR(255) NOT NULL DEFAULT 'entity_list',
    stock_market VARCHAR(50) NOT NULL,
    file_uuid UUID NOT NULL REFERENCES file_list(file_id) ON DELETE CASCADE,
    corporate_id_number VARCHAR(100) NOT NULL,        -- Fix C: Accommodates composite/extended CIN strings
    isin VARCHAR(12) NULL,                            -- Authoritative ISIN extracted from XBRL / Exchange
    nse_symbol VARCHAR(50) NULL,                      -- Authoritative NSE symbol
    bse_scrip_code VARCHAR(20) NULL,                  -- Authoritative BSE scrip code
    company_name VARCHAR(255) NOT NULL,
    address_corporate_office TEXT NULL,
    address_registered_office TEXT NULL,
    contact_person_name VARCHAR(500) NULL,
    contact_person_phone VARCHAR(255) NULL,
    contact_person_email VARCHAR(255) NULL,
    company_email VARCHAR(255) NULL,
    company_telephone VARCHAR(255) NULL,
    website VARCHAR(255) NULL,
    reporting_boundary VARCHAR(50) NULL DEFAULT 'Standalone',
    financial_year_start DATE NULL,
    financial_year_end DATE NULL,
    incorporation_date DATE NULL,
    paid_up_shares_value NUMERIC(20, 4) NULL,

    -- Extracts NIC code ONLY when corporate_id_number matches a structurally valid 21-char MCA CIN
    nic_code VARCHAR(5) GENERATED ALWAYS AS (
        substring(UPPER(TRIM(corporate_id_number)) FROM '^[LU]([0-9]{5})[A-Z]{2}[0-9]{4}[A-Z]{3}[0-9]{6}$')
    ) STORED,
    industry_division VARCHAR(120) NULL,
    industry_sector VARCHAR(100) NULL,                -- Expanded to VARCHAR(100) to match exchange_industry_master.sector
    industry_source VARCHAR(30) NULL,                 -- Audit trail: EXCHANGE_ISIN, EXCHANGE_TICKER, EXCHANGE_CIN, EXCHANGE_NAME, UNCLASSIFIED

    -- Geospatial & Geocoding Fields
    latitude DOUBLE PRECISION NULL,
    longitude DOUBLE PRECISION NULL,
    geocode_status VARCHAR(50) NOT NULL DEFAULT 'PENDING', -- 'PENDING', 'RESOLVED', 'NOT_FOUND', 'ERROR'
    resolved_state VARCHAR(100) NULL,
    resolved_city VARCHAR(100) NULL,

    created_at TIMESTAMPTZ NOT NULL DEFAULT clock_timestamp(),

    -- Fix C: Expressions correctly enclosed in parentheses for valid PostgreSQL GIST syntax
    CONSTRAINT no_overlapping_filings EXCLUDE USING gist (
        corporate_id_number WITH =,
        (COALESCE(reporting_boundary, 'Standalone')) WITH =,
        (daterange(financial_year_start, financial_year_end, '[]')) WITH &&
    ) WHERE (financial_year_start IS NOT NULL AND financial_year_end IS NOT NULL)
);

-- index
CREATE INDEX idx_entity_cin
ON entity_list (corporate_id_number);

CREATE INDEX idx_entity_isin
ON entity_list (isin);

CREATE INDEX idx_entity_nse_symbol
ON entity_list (UPPER(nse_symbol));

CREATE INDEX idx_entity_bse_scrip
ON entity_list (bse_scrip_code);

CREATE INDEX idx_entity_file_uuid
ON entity_list (file_uuid);

CREATE INDEX idx_entity_sector
ON entity_list (industry_sector);

CREATE INDEX idx_entity_industry_source
ON entity_list (industry_source);

CREATE INDEX idx_entity_coords 
ON entity_list (latitude, longitude);

CREATE INDEX idx_entity_geocode_status 
ON entity_list (geocode_status);

-- function - Multi-Key Deterministic Industry Resolution (No legacy NIC-from-CIN fallback)
CREATE OR REPLACE FUNCTION trg_enrich_entity_industry()
RETURNS TRIGGER AS $$
DECLARE
    clean_cin VARCHAR(21);
    clean_isin VARCHAR(12);
    clean_sym VARCHAR(50);
    clean_scrip VARCHAR(20);
    clean_name VARCHAR(255);
    ex RECORD;
BEGIN
    clean_isin  := NULLIF(UPPER(TRIM(NEW.isin)), '');
    clean_sym   := NULLIF(UPPER(TRIM(NEW.nse_symbol)), '');
    clean_scrip := NULLIF(TRIM(NEW.bse_scrip_code), '');
    clean_cin   := substring(UPPER(TRIM(COALESCE(NEW.corporate_id_number, '')))
                   FROM '([LU][0-9]{5}[A-Z]{2}[0-9]{4}[A-Z]{3}[0-9]{6})');

    -- Tier 1: ISIN (Highest accuracy, works for conglomerates & statutory banks)
    IF clean_isin IS NOT NULL THEN
        SELECT sector, industry
        INTO ex
        FROM exchange_industry_master
        WHERE isin = clean_isin
        LIMIT 1;

        IF FOUND THEN
            NEW.industry_sector := ex.sector;
            NEW.industry_division := ex.industry;
            NEW.industry_source := 'EXCHANGE_ISIN';
            RETURN NEW;
        END IF;
    END IF;

    -- Tier 2: Official Exchange Scrip Code (BSE) or Symbol (NSE)
    IF clean_sym IS NOT NULL OR clean_scrip IS NOT NULL THEN
        SELECT sector, industry
        INTO ex
        FROM exchange_industry_master
        WHERE (clean_sym IS NOT NULL AND UPPER(symbol) = clean_sym)
           OR (clean_scrip IS NOT NULL AND scrip_code = clean_scrip)
        LIMIT 1;

        IF FOUND THEN
            NEW.industry_sector := ex.sector;
            NEW.industry_division := ex.industry;
            NEW.industry_source := 'EXCHANGE_TICKER';
            RETURN NEW;
        END IF;
    END IF;

    -- Tier 3: Validated 21-char CIN match against Exchange Master
    IF clean_cin IS NOT NULL THEN
        SELECT sector, industry
        INTO ex
        FROM exchange_industry_master
        WHERE cin = clean_cin
        LIMIT 1;

        IF FOUND THEN
            NEW.industry_sector := ex.sector;
            NEW.industry_division := ex.industry;
            NEW.industry_source := 'EXCHANGE_CIN';
            RETURN NEW;
        END IF;
    END IF;

    -- Tier 4: Normalized Company Name Fuzzy Match (pg_trgm >= 0.85)
    IF NEW.company_name IS NOT NULL AND TRIM(NEW.company_name) <> '' THEN
        clean_name := TRIM(REGEXP_REPLACE(
            REGEXP_REPLACE(UPPER(NEW.company_name), '\m(LIMITED|LTD|PRIVATE|PVT|INDIA|CO|CORP|CORPORATION)\M', '', 'g'),
            '[^A-Z0-9]+', ' ', 'g'
        ));

        IF LENGTH(clean_name) >= 4 THEN
            SELECT sector, industry
            INTO ex
            FROM exchange_industry_master
            WHERE normalized_name % clean_name
              AND similarity(normalized_name, clean_name) >= 0.85
            ORDER BY similarity(normalized_name, clean_name) DESC
            LIMIT 1;

            IF FOUND THEN
                NEW.industry_sector := ex.sector;
                NEW.industry_division := ex.industry;
                NEW.industry_source := 'EXCHANGE_NAME';
                RETURN NEW;
            END IF;
        END IF;
    END IF;

    -- Tier 5: Quarantine instead of guessing from legacy NIC numbers
    NEW.industry_division := 'Pending Exchange Mapping';
    NEW.industry_sector := 'Unclassified';
    NEW.industry_source := 'UNCLASSIFIED';

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- function - Backfill utility to re-resolve unclassified entities after updating exchange_industry_master
CREATE OR REPLACE FUNCTION sync_entity_industries()
RETURNS INTEGER AS $$
DECLARE
    updated_count INTEGER;
BEGIN
    UPDATE entity_list
    SET corporate_id_number = corporate_id_number
    WHERE industry_source IS NULL
       OR industry_source = 'UNCLASSIFIED';

    GET DIAGNOSTICS updated_count = ROW_COUNT;
    RETURN updated_count;
END;
$$ LANGUAGE plpgsql;

-- trigger
CREATE OR REPLACE TRIGGER trg_set_industry
BEFORE INSERT OR UPDATE OF corporate_id_number, isin, nse_symbol, bse_scrip_code, company_name
ON entity_list
FOR EACH ROW
EXECUTE FUNCTION trg_enrich_entity_industry();

CREATE OR REPLACE TRIGGER trg_set_coordinates
BEFORE INSERT OR UPDATE OF address_registered_office, address_corporate_office
ON entity_list
FOR EACH ROW
EXECUTE FUNCTION trg_resolve_entity_coordinates();
